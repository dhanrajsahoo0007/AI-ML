There are n cities. Some of them are connected, while some are not. If city a is connected directly with city b,
and city b is connected directly with city c, then city a is connected indirectly with city c.

A province is a group of directly or indirectly connected cities and no other cities outside of the group.

You are given an n x n matrix isConnected where isConnected\[i\]\[j\] = 1 if the ith city and the jth city are directly connected,
and isConnected\[i\]\[j\] = 0 otherwise.

Return the total number of provinces.

## Examples

````text
Example 1:
Input: isConnected = [[1,1,0],[1,1,0],[0,0,1]]
Output: 2
````

## Constraints

````text
- 1 <= n <= 200
- isConnected[i][j] is 1 or 0.
- isConnected[i][i] == 1
````

## Approach

**Pattern:** Union-Find / Disjoint Set Union

1. Initialize Union-Find with n nodes.
2. For each pair i &lt; j with isConnected\[i\]\[j\] == 1, union(i, j).
3. Return the number of connected components tracked by the DSU.

## Complexity

**Time:** O(n^2 \* α(n))

**Space:** O(n)

## Solution

````python
from typing import List


class UnionFind:
    def __init__(self, size: int) -> None:
        self.parent = list(range(size))
        self.rank = [0] * size
        self.components = size

    def find(self, node: int) -> int:
        if self.parent[node] != node:
            self.parent[node] = self.find(self.parent[node])
        return self.parent[node]

    def union_by_rank(self, source: int, target: int) -> bool:
        psource = self.find(source)
        ptarget = self.find(target)
        if psource == ptarget:
            return False
        if self.rank[psource] > self.rank[ptarget]:
            self.parent[ptarget] = psource
        elif self.rank[psource] < self.rank[ptarget]:
            self.parent[psource] = ptarget
        else:
            self.parent[ptarget] = psource
            self.rank[psource] += 1
        self.components -= 1
        return True


class Solution:
    def findCircleNum(self, isConnected: List[List[int]]) -> int:
        n = len(isConnected)
        union_finder = UnionFind(n)
        for i in range(n):
            for j in range(i + 1, n):
                if isConnected[i][j] == 1:
                    union_finder.union_by_rank(i, j)
        return union_finder.components
````

## Another version

From 17. Graphs/03. DFS Story Based/01. Introduction to Graph Traversal.

### Approach

**Pattern:** Connected components via DFS
Each DFS/BFS from an unvisited city marks one province; count DFS starts.

1. Track visited cities.
2. For each unvisited i, run DFS on the matrix row i to mark all reachable cities.
3. Increment province count after each DFS launch.

### Complexity

**Time:** O(n^2) — each cell of the matrix may be inspected when exploring neighbors.

**Space:** O(n) — visited array and recursion stack.

### Solution

````python
from typing import List


class Solution:
    def findCircleNum(self, isConnected: List[List[int]]) -> int:
        n = len(isConnected)
        visited = [False] * n
        provinces = 0

        def dfs(u: int) -> None:
            visited[u] = True
            for v in range(n):
                if isConnected[u][v] == 1 and not visited[v]:
                    dfs(v)

        for i in range(n):
            if not visited[i]:
                dfs(i)
                provinces += 1
        return provinces
````

## Another version

From 17. Graphs/04. BFS Story Based/01. Introduction to Graph Traversal with BFS.

### Approach

**Pattern:** Graph BFS (connected components)

### Complexity

**Time:** O(n^2) for both approaches — scanning the matrix to build edges or as neighbors.

**Space:** O(n^2) for adjacency list in approach 1; O(n) for visited and queue.

Original section headers preserved in comments below: "Approach - 1" and "Approach - 2".

### Solution

````python
from typing import List
from collections import deque


class Solution:
    # ---------- Approach - 1: Creating the adjacency list from the matrix then solving it ----------

    def create_adjacency_list(self, isConnected: List[List[int]]) -> List[List[int]]:
        n = len(isConnected)
        adj: List[List[int]] = [[] for _ in range(n)]
        for i in range(n):
            for j in range(n):
                if isConnected[i][j] == 1 and i != j:
                    adj[i].append(j)
        return adj

    def _bfs_adj(self, adj: List[List[int]], u: int, visited: List[bool]) -> None:
        queue = deque([u])
        visited[u] = True

        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    queue.append(v)

    def findCircleNum(self, isConnected: List[List[int]]) -> int:
        n = len(isConnected)
        adj = self.create_adjacency_list(isConnected)
        visited = [False] * n
        provinces = 0

        for i in range(n):
            if not visited[i]:
                self._bfs_adj(adj, i, visited)
                provinces += 1

        return provinces

    # ---------- Approach - 2: Directly iterating through the input matrix ----------

    def _bfs_matrix(
        self, isConnected: List[List[int]], u: int, visited: List[bool], result: List[int]
    ) -> None:
        queue = deque([u])
        visited[u] = True

        while queue:
            u = queue.popleft()
            result.append(u)
            for v in range(len(isConnected)):
                if isConnected[u][v] == 1 and not visited[v]:
                    visited[v] = True
                    queue.append(v)

    def findCircleNumMatrix(self, isConnected: List[List[int]]) -> int:
        n = len(isConnected)
        visited = [False] * n
        provinces = 0

        for i in range(n):
            if not visited[i]:
                result: List[int] = []
                self._bfs_matrix(isConnected, i, visited, result)
                provinces += 1

        return provinces
````
