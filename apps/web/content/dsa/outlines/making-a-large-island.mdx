You are given an n x n binary matrix grid. You are allowed to change at most one 0 to be 1.
Return the size of the largest island in grid after applying this operation.
An island is a 4-directionally connected group of 1s.

## Examples

````text
Example 1:
    Input: grid = [[1,0],[0,1]]
    Output: 3
Example 2:
    Input: grid = [[1,1],[1,0]]
    Output: 4
Example 3:
    Input: grid = [[1,1],[1,1]]
    Output: 4
````

## Constraints

````text
- n == grid.length == grid[i].length
- 1 <= n <= 500
- grid[i][j] is either 0 or 1.
````

## Approach

**Pattern:** Union-Find / Disjoint Set Union on grid cells

1. DSU over n\*n cells; union adjacent 1s.
2. Track max existing island size from component sizes at roots.
3. For each 0, sum sizes of distinct neighboring components plus one; take max.

## Complexity

**Time:** O(n^2), where n is the side length of the grid

**Space:** O(n^2) for the DSU structure

This solution uses a Disjoint Set Union (DSU) data structure to efficiently connect and measure islands.

## Solution

````python
from typing import List


class DSU:
    def __init__(self, n: int) -> None:
        self.parent = list(range(n))
        self.size = [1] * n
        self.number_of_components = n

    def find(self, x: int) -> int:
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x: int, y: int) -> None:
        x_parent, y_parent = self.find(x), self.find(y)
        if x_parent == y_parent:
            return

        if self.size[x_parent] > self.size[y_parent]:
            self.parent[y_parent] = x_parent
            self.size[x_parent] += self.size[y_parent]
        else:
            self.parent[x_parent] = y_parent
            self.size[y_parent] += self.size[x_parent]

        self.number_of_components -= 1


class Solution:
    def largestIsland(self, grid: List[List[int]]) -> int:
        n = len(grid)
        dsu = DSU(n * n)

        directions = [(0, 1), (1, 0), (0, -1), (-1, 0)]

        def is_valid(x: int, y: int) -> bool:
            return 0 <= x < n and 0 <= y < n

        for i in range(n):
            for j in range(n):
                if grid[i][j] == 1:
                    for dx, dy in directions:
                        ni, nj = i + dx, j + dy
                        if is_valid(ni, nj) and grid[ni][nj] == 1:
                            dsu.union(i * n + j, ni * n + nj)

        max_size = max(dsu.size) if n else 0

        for i in range(n):
            for j in range(n):
                if grid[i][j] == 0:
                    neighbor_roots: set[int] = set()
                    for dx, dy in directions:
                        ni, nj = i + dx, j + dy
                        if is_valid(ni, nj) and grid[ni][nj] == 1:
                            neighbor_roots.add(dsu.find(ni * n + nj))

                    size = 1
                    for root in neighbor_roots:
                        size += dsu.size[root]

                    max_size = max(max_size, size)

        return max_size
````

## Another version

From 17. Graphs/13. Flood Fill.

### Approach

**Pattern:** Component coloring + neighbor merge
Label each original island with an id and its area, then evaluate each 0 by summing distinct neighbor ids.

1. DFS paint each island with an id &gt;=2 while recording area sizes.
2. Track the maximum existing island size.
3. For each 0 cell, add 1 plus areas of unique neighboring component ids.

### Complexity

**Time:** O(n^2)

**Space:** O(n^2) for recursion and grid labels.

### Solution

````python
from typing import List


class Solution:
    def largestIsland(self, grid: List[List[int]]) -> int:
        n = len(grid)
        dirs = ((1, 0), (-1, 0), (0, 1), (0, -1))
        index = 2
        area: dict[int, int] = {}

        def dfs(r: int, c: int, color: int) -> int:
            grid[r][c] = color
            cnt = 1
            for dr, dc in dirs:
                nr, nc = r + dr, c + dc
                if 0 <= nr < n and 0 <= nc < n and grid[nr][nc] == 1:
                    cnt += dfs(nr, nc, color)
            return cnt

        for i in range(n):
            for j in range(n):
                if grid[i][j] == 1:
                    area[index] = dfs(i, j, index)
                    index += 1

        ans = max(area.values(), default=0)
        for i in range(n):
            for j in range(n):
                if grid[i][j] != 0:
                    continue
                seen: set[int] = set()
                cur = 1
                for dr, dc in dirs:
                    ni, nj = i + dr, j + dc
                    if 0 <= ni < n and 0 <= nj < n and grid[ni][nj] >= 2:
                        seen.add(grid[ni][nj])
                for comp in seen:
                    cur += area[comp]
                ans = max(ans, cur)
        return ans
````
