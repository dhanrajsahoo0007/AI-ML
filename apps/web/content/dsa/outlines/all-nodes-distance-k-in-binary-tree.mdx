Given the root of a binary tree, a target node (by reference), and an integer k, return values of all nodes
at distance k from the target (edges). Answer order is arbitrary.

## Examples

````text
Example 1:
    Tree [3,5,1,6,2,0,8,None,None,7,4], target = node(5), k = 2
    Output: contains {7,4,1} in any order
Example 2:
    Tree [1], target = node(1), k = 3
    Output: []
````

## Constraints

````text
- Nodes in [1, 500], unique values, target exists, 0 <= k <= 1000.
````

## Approach

**Pattern:** Graph BFS after parent links
Treat tree as undirected graph: add parent pointers, BFS k layers from target.

1. DFS annotate parent for every node.
2. BFS from target with visited set; when depth equals k, collect queue values.

## Complexity

**Time:** O(n)

**Space:** O(n) for parent map, queue, and visited.

## Solution

````python
from collections import deque
from typing import List, Optional, Set


class TreeNode:
    def __init__(self, val: int = 0, left: Optional["TreeNode"] = None, right: Optional["TreeNode"] = None):
        self.val = val
        self.left = left
        self.right = right


def build_tree_and_find(values: List[Optional[int]], target_val: int) -> tuple[Optional[TreeNode], TreeNode]:
    if not values or values[0] is None:
        raise ValueError("empty tree")
    root = TreeNode(values[0])
    q: deque[TreeNode] = deque([root])
    target_node = root if values[0] == target_val else None
    i = 1
    while q and i < len(values):
        node = q.popleft()
        if i < len(values) and values[i] is not None:
            node.left = TreeNode(values[i])
            if values[i] == target_val:
                target_node = node.left
            q.append(node.left)
        i += 1
        if i < len(values) and values[i] is not None:
            node.right = TreeNode(values[i])
            if values[i] == target_val:
                target_node = node.right
            q.append(node.right)
        i += 1
    if target_node is None:
        raise ValueError("target not in tree")
    return root, target_node


class Solution:
    def distanceK(self, root: Optional[TreeNode], target: TreeNode, k: int) -> List[int]:
        parent: dict[TreeNode, Optional[TreeNode]] = {}

        def attach(node: Optional[TreeNode], par: Optional[TreeNode]) -> None:
            if not node:
                return
            parent[node] = par
            attach(node.left, node)
            attach(node.right, node)

        attach(root, None)
        q: deque[TreeNode] = deque([target])
        seen: Set[TreeNode] = {target}
        dist = 0
        while q:
            if dist == k:
                return [n.val for n in q]
            for _ in range(len(q)):
                cur = q.popleft()
                for nb in (cur.left, cur.right, parent.get(cur)):
                    if nb and nb not in seen:
                        seen.add(nb)
                        q.append(nb)
            dist += 1
        return []
````
