Parse log lines Start(time, reqId) and Finish(time, reqId), pair each request's start/end,
and produce lines sorted by finish time ascending. Handle out-of-order logs; detect invalid
finish-before-start.

## Examples

````text
Example 1:
Logs with mixed order -> output sorted by end time with readable lines.
````

## Constraints

````text
- Each reqId has at most one Start and one Finish when complete; missing Finish yields 'unknown'.
````

<PremiumBreak />

## Approach

**Pattern:** Hash map aggregation + sort
Store start/finish per id while streaming; build records then sort by end key.

1. Regex-parse timestamps and ids from each line.
2. Merge into a dict of partial records; flag finish before start as error when detected.
3. Sort completed rows by finish time; format title-case id in output.

## Complexity

**Time:** O(L log L) for L requests when sorting.

**Space:** O(L) for stored intervals.

## Solution

````python
import re
from dataclasses import dataclass
from typing import Dict, List, Optional, Tuple


@dataclass
class _Req:
    start: Optional[int] = None
    finish: Optional[int] = None


class GoogleDocAccessLog:
    _pat = re.compile(r"^(Start|Finish)\((\d+),\s*([^)]+)\)\s*$", re.I)

    @classmethod
    def _parse(cls, line: str) -> Tuple[str, int, str]:
        m = cls._pat.match(line.strip())
        if not m:
            raise ValueError(f"bad log line: {line!r}")
        kind, ts, rid = m.group(1).lower(), int(m.group(2)), m.group(3).strip()
        return kind, ts, rid

    @classmethod
    def process(cls, lines: List[str]) -> str:
        data: Dict[str, _Req] = {}
        errors: List[str] = []
        for raw in lines:
            if not raw.strip():
                continue
            kind, ts, rid = cls._parse(raw)
            rec = data.setdefault(rid, _Req())
            if kind == "start":
                if rec.start is not None and rec.start != ts:
                    errors.append(f"duplicate start for {rid}")
                rec.start = ts
                if rec.finish is not None and rec.finish < ts:
                    errors.append(f"start after finish for {rid}")
            else:
                if rec.finish is not None:
                    errors.append(f"duplicate finish for {rid}")
                rec.finish = ts
                if rec.start is not None and ts < rec.start:
                    errors.append(f"finish before start for {rid}")

        if errors:
            return "Error: " + "; ".join(errors)

        rows: List[Tuple[int, str, int, int]] = []
        for rid, rec in data.items():
            if rec.finish is None:
                continue
            s = rec.start if rec.start is not None else -1
            rows.append((rec.finish, rid, s, rec.finish))
        rows.sort(key=lambda x: x[0])

        out_lines: List[str] = []
        for _, rid, s, e in rows:
            label = rid[:1].upper() + rid[1:] if rid else rid
            ss = "unknown" if s < 0 else str(s)
            out_lines.append(f"{label} started at {ss} and ended at {e}")
        for rid, rec in sorted(data.items()):
            if rec.finish is None and rec.start is not None:
                label = rid[:1].upper() + rid[1:]
                out_lines.append(f"{label} started at {rec.start} and ended at unknown")
        return "\n".join(out_lines)
````
