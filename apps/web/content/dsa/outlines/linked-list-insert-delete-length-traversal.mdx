Implement a singly linked list supporting insert at beginning/end, insert and delete at index,
delete first node with a given value, length, and traversal (as values for testing).

## Examples

````text
Example 1:
Input: build [1,2,3], insert 0 at beginning, delete value 2
Output: list order [0,1,3], length 3
````

## Constraints

````text
- Values are integers; indices are 0-based; invalid index operations leave list unchanged.
````

<PremiumBreak />

## Approach

**Pattern:** Sentinel / pointer manipulation
Use a clear head pointer and careful relinking for O(1) or O(n) updates.

1. Represent nodes as ListNode(val, next).
2. For insert at index, walk to predecessor or splice at head.
3. For delete at index, unlink predecessor.next.
4. Count nodes for length; collect values for traversal snapshot.

## Complexity

**Time:** O(n) per operation that scans the list; O(1) for insert at head when head is known.

**Space:** O(1) extra aside from output lists for tests.

## Solution

````python
from __future__ import annotations

from typing import List, Optional


class ListNode:
    def __init__(self, val: int = 0, next: Optional[ListNode] = None):
        self.val = val
        self.next = next


def list_to_linked_list(values: List[int]) -> Optional[ListNode]:
    dummy = ListNode(0)
    cur = dummy
    for v in values:
        cur.next = ListNode(v)
        cur = cur.next
    return dummy.next


def linked_list_to_list(head: Optional[ListNode]) -> List[int]:
    out: List[int] = []
    while head:
        out.append(head.val)
        head = head.next
    return out


class Solution:
    def __init__(self) -> None:
        self.head: Optional[ListNode] = None

    def insert_at_beginning(self, val: int) -> None:
        self.head = ListNode(val, self.head)

    def insert_at_end(self, val: int) -> None:
        if self.head is None:
            self.head = ListNode(val)
            return
        cur = self.head
        while cur.next:
            cur = cur.next
        cur.next = ListNode(val)

    def delete_value(self, val: int) -> None:
        while self.head and self.head.val == val:
            self.head = self.head.next
        cur = self.head
        while cur and cur.next:
            if cur.next.val == val:
                cur.next = cur.next.next
            else:
                cur = cur.next

    def length(self) -> int:
        n = 0
        cur = self.head
        while cur:
            n += 1
            cur = cur.next
        return n

    def values(self) -> List[int]:
        return linked_list_to_list(self.head)

    def insert_at_index(self, val: int, index: int) -> None:
        if index < 0:
            return
        if index == 0:
            self.insert_at_beginning(val)
            return
        cur = self.head
        for _ in range(index - 1):
            if cur is None:
                return
            cur = cur.next
        if cur is None:
            return
        cur.next = ListNode(val, cur.next)

    def delete_at_index(self, index: int) -> None:
        if self.head is None or index < 0:
            return
        if index == 0:
            self.head = self.head.next
            return
        cur = self.head
        for _ in range(index - 1):
            if cur is None or cur.next is None:
                return
            cur = cur.next
        if cur.next is None:
            return
        cur.next = cur.next.next
````

## Tests from the notes

````python
if __name__ == "__main__":
    sol = Solution()
    sol.insert_at_end(1)
    sol.insert_at_end(2)
    sol.insert_at_beginning(0)
    sol.insert_at_end(3)
    assert sol.values() == [0, 1, 2, 3]
    assert sol.length() == 4

    sol.delete_value(2)
    assert sol.values() == [0, 1, 3]
    assert sol.length() == 3

    sol2 = Solution()
    sol2.head = list_to_linked_list([10, 20, 30])
    sol2.insert_at_index(15, 1)
    assert sol2.values() == [10, 15, 20, 30]

    sol3 = Solution()
    sol3.head = list_to_linked_list([1, 2, 3])
    sol3.delete_at_index(1)
    assert sol3.values() == [1, 3]

    sol4 = Solution()
    sol4.head = list_to_linked_list([5])
    sol4.delete_at_index(0)
    assert sol4.head is None

    print("\nAll test cases passed!")
````
