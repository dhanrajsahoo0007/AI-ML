Given an integer array nums, return all unique triplets \[nums\[i\], nums\[j\], nums\[k\]\] such that
i, j, and k are distinct indices and their values sum to zero.

## Examples

````text
Example 1:
    Input: nums = [-1, 0, 1, 2, -1, -4]
    Output: [[-1, -1, 2], [-1, 0, 1]]
Example 2:
    Input: nums = [0, 1, 1]
    Output: []
````

## Constraints

````text
- 3 <= len(nums) <= 3000
- -10^5 <= nums[i] <= 10^5
````

## Approach

**Pattern:** Sort + two pointers
Fix the first element, then run a sorted two-sum on the remainder to find pairs that complete zero.

1. Sort nums to enable skipping duplicates and two-pointer scanning.
2. For each index i up to n - 3, skip duplicate values of nums\[i\].
3. Use two pointers on the subarray after i to find pairs summing to -nums\[i\], skipping duplicate pairs.
4. Collect every unique triplet found.

## Solution

from typing import List


class Solution:
    def threeSum(self, nums: List[int]) -> List[List[int]]:
        nums.sort()
        n = len(nums)
        result: List[List[int]] = []
        for i in range(n - 2):
            if i > 0 and nums[i] == nums[i - 1]:
                continue
            target = -nums[i]
            left, right = i + 1, n - 1
            while left < right:
                current = nums[left] + nums[right]
                if current > target:
                    right -= 1
                elif current < target:
                    left += 1
                else:
                    result.append([nums[i], nums[left], nums[right]])
                    left += 1
                    right -= 1
                    while left < right and nums[left] == nums[left - 1]:
                        left += 1
                    while left < right and nums[right] == nums[right + 1]:
                        right -= 1
        return result


class ThreeSumUsingTwoSumHelper:
    """
    Alternate structure from the archived solution: extract the inner two-pointer pass into a
    dedicated helper while the outer loop fixes the first element of each triplet.
    """

    def twoSum(self, nums: List[int], k: int, result: List[List[int]], target: int) -> None:
        i, j = k, len(nums) - 1
        while i < j:
            current_sum = nums[i] + nums[j]
            if current_sum > target:
                j -= 1  # Sum too large, decrease right pointer
            elif current_sum < target:
                i += 1  # Sum too small, increase left pointer
            else:
                result.append([-target, nums[i], nums[j]])
                while i < j and nums[i] == nums[i + 1]:
                    i += 1
                while i < j and nums[j] == nums[j - 1]:
                    j -= 1
                i += 1
                j -= 1

    def threeSum(self, nums: List[int]) -> List[List[int]]:
        if len(nums) < 3:
            return []
        result: List[List[int]] = []
        nums.sort()
        for i in range(len(nums) - 2):
            if i > 0 and nums[i] == nums[i - 1]:
                continue
            self.twoSum(nums, i + 1, result, -nums[i])
        return result


def _triplets_key(trips: List[List[int]]) -> List[tuple[int, ...]]:
    return sorted(tuple(sorted(t)) for t in trips)
